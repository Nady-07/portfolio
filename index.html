// Dynamic YoY Net Revenue & Margin Telemetry
Net Revenue YoY % = 
VAR CurrentRev = [Total Net Revenue]
VAR PriorYearRev = 
    CALCULATE(
        [Total Net Revenue],
        SAMEPERIODLASTYEAR( 'Dim_Date'[Date] )
    )
RETURN
    DIVIDE( CurrentRev - PriorYearRev, PriorYearRev, 0 )

Gold_Layer_Analytics.sql · T-SQL 
WITH CustomerRFM AS (
    SELECT
        customer_id,
        SUM(net_amount) AS lifetime_value,
        DENSE_RANK() OVER (
            PARTITION BY region 
            ORDER BY SUM(net_amount) DESC
        ) AS regional_rank
    FROM gold.fact_sales
    GROUP BY customer_id, region
)
SELECT * FROM CustomerRFM WHERE regional_rank <= 10;
